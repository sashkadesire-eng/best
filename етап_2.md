"""Каталог: тарифи, дошки, локації, пляжі. Усі тексти — parler (EN/ES)."""
from django.db import models
from django.utils.translation import get_language
from parler.models import TranslatableModel, TranslatedFields


class Tariff(TranslatableModel):
    code = models.SlugField(unique=True)  # 2h / 4h / 8h / multi
    duration_hours = models.PositiveIntegerField()
    price_per_board = models.PositiveIntegerField()
    is_popular = models.BooleanField(default=False)
    order = models.PositiveIntegerField(default=0)
    is_active = models.BooleanField(default=True)
    # Особливості картки (en/es списки) — динамічно з адмінки
    features = models.JSONField(default=dict, blank=True)

    translations = TranslatedFields(
        name=models.CharField(max_length=120),
        badge=models.CharField(max_length=120, blank=True),
    )

    class Meta:
        ordering = ["order"]

    def __str__(self):
        return f"{self.code} · €{self.price_per_board}"


class Board(TranslatableModel):
    code = models.SlugField(unique=True)  # convertible-100 / wattsup-102 / wattsup-silver-116
    name = models.CharField(max_length=120)  # назви інтернаціональні
    capacity = models.CharField(max_length=120)
    length = models.CharField(max_length=60)
    image = models.ImageField(upload_to="boards/", blank=True)
    image_url = models.URLField(blank=True)
    order = models.PositiveIntegerField(default=0)
    is_active = models.BooleanField(default=True)
    features = models.JSONField(default=dict, blank=True)  # {"en": [...], "es": [...]}

    translations = TranslatedFields(description=models.TextField(blank=True))

    class Meta:
        ordering = ["order"]

    def __str__(self):
        return self.name


class Location(TranslatableModel):
    code = models.SlugField(unique=True)  # benidorm / altea / villajoyosa / calpe / alicante
    name = models.CharField(max_length=80)
    province = models.CharField(max_length=120, default="Alicante, Costa Blanca")
    lat = models.FloatField(null=True, blank=True)
    lng = models.FloatField(null=True, blank=True)
    image = models.ImageField(upload_to="locations/", blank=True)
    image_url = models.URLField(blank=True)
    order = models.PositiveIntegerField(default=0)
    is_active = models.BooleanField(default=True)

    translations = TranslatedFields(
        description=models.TextField(blank=True),
        long_description=models.TextField(blank=True),
        meta_title=models.CharField(max_length=200, blank=True),
        meta_description=models.CharField(max_length=320, blank=True),
        popular_spots=models.JSONField(default=list, blank=True),
        # SEO-наратив (LOCATION_SEO_NARRATIVES)
        intro=models.TextField(blank=True),
        summary=models.TextField(blank=True),
    )

    class Meta:
        ordering = ["order"]

    def __str__(self):
        return self.name

    def get_absolute_url(self):
        """Локалізований URL: /en/sup-rental-X/ ↔ /es/alquiler-sup-X/ (карта URL v3)."""
        lang = get_language()
        prefix = "alquiler-sup" if lang == "es" else "sup-rental"
        return f"/{lang}/{prefix}-{self.code}/"

    @property
    def beaches_count(self):
        return self.beaches.count()


class Beach(models.Model):
    location = models.ForeignKey(Location, related_name="beaches", on_delete=models.CASCADE)
    name = models.CharField(max_length=120)  # власні назви — спільні для обох мов
    order = models.PositiveIntegerField(default=0)

    class Meta:
        ordering = ["order"]
        verbose_name_plural = "beaches"

    def __str__(self):
        return f"{self.location_id}: {self.name}"


class LocationHighlight(TranslatableModel):
    """SEO-наратив: ключові особливості локації (intro → highlights → summary)."""

    location = models.ForeignKey(Location, related_name="highlights", on_delete=models.CASCADE)
    order = models.PositiveIntegerField(default=0)

    translations = TranslatedFields(
        title=models.CharField(max_length=200),
        text=models.TextField(blank=True),
    )

    class Meta:
        ordering = ["order"]

    def __str__(self):
        return self.title or f"Highlight #{self.pk}"

"""Лендінги локацій. URL-карта v3: /en/sup-rental-X/ ↔ /es/alquiler-sup-X/.

Суворий резолвер: /es/sup-rental-benidorm/ → 404 (префікс не відповідає мові).
"""
from django.http import Http404
from django.shortcuts import get_object_or_404
from django.utils.translation import get_language
from django.views.generic import DetailView

from catalog.models import Board, Location, Tariff


class LocationDetailView(DetailView):
    model = Location
    context_object_name = "location"
    template_name = "catalog/location.html"

    def get_object(self, queryset=None):
        lang = get_language()
        expected_prefix = "alquiler-sup" if lang == "es" else "sup-rental"
        if self.kwargs["prefix"] != expected_prefix:
            raise Http404("Невірний мовний префікс URL")
        return get_object_or_404(Location, code=self.kwargs["city"], is_active=True)

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        loc = self.object
        ctx["tariffs"] = Tariff.objects.filter(is_active=True)
        ctx["boards"] = Board.objects.filter(is_active=True)
        ctx["other_locations"] = Location.objects.filter(is_active=True).exclude(pk=loc.pk)
        ctx["meta_title"] = loc.meta_title
        ctx["meta_description"] = loc.meta_description
        # lang_urls для перемикача мов та hreflang (ADR-0007)
        ctx["lang_urls"] = {
            "en": f"/en/sup-rental-{loc.code}/",
            "es": f"/es/alquiler-sup-{loc.code}/",
        }
        return ctx
EOF

from django import template

register = template.Library()


@register.filter
def get_item_lang(mapping, lang):
    """JSON-словник {"en": ..., "es": ...} → значення активною мовою (fallback en)."""
    if isinstance(mapping, dict):
        return mapping.get(lang) or mapping.get("en") or []
    return []
EOF

def set_tr(obj, fields_by_lang):
    for lang, fields in fields_by_lang.items():
        obj.set_current_language(lang)
        for key, value in fields.items():
            setattr(obj, key, value)
    obj.save()
    return obj


def render_blocks(blocks, lang):
    """Структуровані блоки → HTML."""
    out = []
    for block in blocks:
        kind = block[0]
        if kind == "h2":
            out.append(f"<h2>{block[1][lang]}</h2>")
        elif kind == "p":
            out.append(f"<p>{block[1][lang]}</p>")
        elif kind == "ul":
            items = "".join(f"<li>{item[lang]}</li>" for item in block[1])
            out.append(f"<ul>{items}</ul>")
        elif kind == "table":
            headers = "".join(f"<th>{h}</th>" for h in block[1])
            rows = "".join("<tr>" + "".join(f"<td>{c}</td>" for c in row) + "</tr>" for row in block[2])
            out.append(f"<table><thead><tr>{headers}</tr></thead><tbody>{rows}</tbody></table>")
    return "\n".join(out)


def terms_tariff_table(lang):
    """Таблиця тарифів ДИНАМІЧНО з БД (паритет із mockData, баг #18 закрито)."""
    headers = {"en": ["Tariff", "Duration", "Price per Board", "Refundable Deposit"],

    rows = []
    for t in Tariff.objects.filter(is_active=True):
        t.set_current_language(lang)
        duration = f"{t.duration_hours}h" if t.code != "multi" else f"{t.duration_hours}h / day"

                    f"<td><strong>€{t.price_per_board}</strong>{' / day' if t.code == 'multi' else ''}</td>"
                    f"<td>€150 / rental</td></tr>")
    th = "".join(f"<th>{h}</th>" for h in headers[lang])
    return f"<table><thead><tr>{th}</tr></thead><tbody>{''.join(rows)}</tbody></table>"

            set_tr(obj, {"en": {"name": name["en"], "description": desc["en"]},

        for i, (title, detail) in enumerate(MARITIME_LAWS, start=1):

            set_tr(obj, {"en": {"title": title["en"], "detail": detail["en"]},

    def seed_content(self, locations):
        for i, g in enumerate(GALLERY, start=1):
            code, loc_code, cat, img, title, desc, spot = g
            obj, _ = GalleryItem.objects.get_or_create(
                code=code, defaults=dict(location=locations[loc_code], category=cat, image_url=img, order=i * 10),
            )
            set_tr(obj, {"en": {"title": title["en"], "description": desc["en"], "spot_guide": spot["en"]},
                         "es": {"title": title["es"], "description": desc["es"], "spot_guide": spot["es"]}})

        for i, (cat, q, a) in enumerate(FAQS, start=1):
            obj, _ = FAQItem.objects.get_or_create(category=cat, order=i * 10)
            set_tr(obj, {"en": {"question": q["en"], "answer": a["en"]},
                         "es": {"question": q["es"], "answer": a["es"]}})

        for w in WEATHER:
            loc_code, name, temp, water, wind, direction, wave, status = w
            WeatherSpot.objects.update_or_create(
                location=locations[loc_code],
                defaults=dict(name=name, temp=temp, water_temp=water, wind_knots=wind,
                              wind_direction=direction, wave_height=wave, status=status),

            "cookie-policy": lambda lang: render_blocks(COOKIES, lang),
            "legal-notice": lambda lang: render_blocks(LEGAL_NOTICE, lang),
            "terms-and-safety": lambda lang: (
                render_blocks(TERMS_SECTIONS_HEAD, lang)
                + terms_tariff_table(lang)
                + render_blocks(TERMS_AFTER_TARIFFS, lang)
            ),
        }
        for en_slug, builder in pages.items():
            page = Page.objects.filter(translations__language_code="en", translations__slug=en_slug).first()
            if page is None:
                self.stdout.write(self.style.WARNING(f"⚠️  Сторінку {en_slug} не знайдено (запустіть seed_cms)"))
                continue
            section = page.sections.filter(section_type="richtext").first()
            if section is None:
                section = Section.objects.create(page=page, section_type="richtext", order=10)
            section.set_current_language("en")
            section.body = builder("en")
            section.set_current_language("es")
            section.body = builder("es")
            section.save()
