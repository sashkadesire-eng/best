# ── 01. Виправлення: core_filters для шаблонів із get_item_lang ────────────
python - << 'PY'
import pathlib, re
fixed = []
for p in pathlib.Path("templates").rglob("*.html"):
    text = p.read_text(encoding="utf-8")
    if "get_item_lang" not in text:
        continue
    loads = re.findall(r"{%\s*load\s+([^%]+)%}", text)
    if any("core_filters" in l for l in loads):
        continue
    new = re.sub(r"{%\s*load\s+([^%]+)%}",
                 lambda m: "{% load " + m.group(1).strip() + " core_filters %}",
                 text, count=1)
    p.write_text(new, encoding="utf-8")
    fixed.append(str(p))
print("✅ core_filters додано: " + (", ".join(fixed) if fixed else "—"))
PY

"""JSON-LD-граф лендінгу локації (1:1 з schemaMarkup.ts, ціни — динамічно з БД).

BreadcrumbList + Product/AggregateOffer + FAQPage. Google читає структуровані
дані → розширені снипети, краще розуміння географії та цін.
"""
import json

from django.conf import settings

from content.models import FAQItem


def build_location_jsonld(location, tariffs, lang):
    site_url = settings.SITE_URL.rstrip("/")
    prices = [t.price_per_board for t in tariffs]

    # FAQPage — перші 3 активні питання активною мовою
    main_entity = []
    for faq in FAQItem.objects.filter(is_active=True)[:3]:
        faq.set_current_language(lang)
        main_entity.append(
            {
                "@type": "Question",
                "name": faq.question,
                "acceptedAnswer": {"@type": "Answer", "text": faq.answer},
            }
        )

    product_name = (
        f"Alquiler de SUP en {location.name}"
        if lang == "es"
        else f"SUP Board Rental in {location.name}"
    )

    image = ""
    if location.image_url:
        image = location.image_url
    elif location.image:
        try:
            image = location.image.url
        except ValueError:
            image = ""

    product = {
        "@type": "Product",
        "name": product_name,
        "description": location.meta_description,
        "brand": {"@type": "Brand", "name": "TUYYO.CLUB"},
        "offers": {
            "@type": "AggregateOffer",
            "priceCurrency": "EUR",
            "lowPrice": f"{min(prices):.2f}" if prices else "35.00",
            "highPrice": f"{max(prices):.2f}" if prices else "50.00",
            "offerCount": len(prices),
            "priceValidUntil": "2026-12-31",
            "availability": "https://schema.org/InStock",
        },
        "aggregateRating": {
            "@type": "AggregateRating",
            "ratingValue": "4.9",
            "reviewCount": "46",
        },
    }
    if image:
        product["image"] = image

    graph = {
        "@context": "https://schema.org",
        "@graph": [
            {
                "@type": "BreadcrumbList",
                "itemListElement": [
                    {
                        "@type": "ListItem",
                        "position": 1,
                        "name": "Home" if lang == "en" else "Inicio",
                        "item": f"{site_url}/{lang}/",
                    },
                    {
                        "@type": "ListItem",
                        "position": 2,
                        "name": "Costa Blanca",
                        "item": f"{site_url}/{lang}/#locations",
                    },
                    {
                        "@type": "ListItem",
                        "position": 3,
                        "name": location.name,
                        "item": f"{site_url}{location.get_absolute_url()}",
                    },
                ],
            },
            product,
            {"@type": "FAQPage", "mainEntity": main_entity},
        ],
    }
    # Безпечний JSON для <script> (екрануємо "</")
    return json.dumps(graph, ensure_ascii=False).replace("</", "<\\/")
EOF

# ── 04. catalog/views.py: DetailView + GuideTabView (суворий резолвер) ─────
cat > catalog/views.py << 'EOF'
"""Лендінги локацій + HTMX-вкладки гайда. Карта URL v3:
/en/sup-rental-X/ ↔ /es/alquiler-sup-X/. Суворий резолвер: префікс має
відповідати активній мові, інакше 404 (SEO: унікальні URL для кожної мови).
"""
from django.http import Http404
from django.shortcuts import get_object_or_404
from django.utils.translation import get_language
from django.views.generic import DetailView, TemplateView

from catalog.models import Board, Location, Tariff
from catalog.seo import build_location_jsonld
from content.models import WeatherSpot
from guides.models import BeachSpotIntel, GuideOverview, MaritimeLaw, SupRoute, WindWindow


def resolve_location(prefix, city):
    """Суворий мовний резолвер: /es/sup-rental-X/ → 404."""
    lang = get_language()
    expected = "alquiler-sup" if lang == "es" else "sup-rental"
    if prefix != expected:
        raise Http404("Невірний мовний префікс URL")
    return get_object_or_404(Location, code=city, is_active=True)


class LocationDetailView(DetailView):
    model = Location
    context_object_name = "location"
    template_name = "catalog/location.html"

    def get_object(self, queryset=None):
        return resolve_location(self.kwargs["prefix"], self.kwargs["city"])

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        loc = self.object
        tariffs = Tariff.objects.filter(is_active=True)
        ctx["tariffs"] = tariffs
        ctx["boards"] = Board.objects.filter(is_active=True)
        ctx["other_locations"] = Location.objects.filter(is_active=True).exclude(pk=loc.pk)
        ctx["meta_title"] = loc.meta_title
        ctx["meta_description"] = loc.meta_description
        ctx["lang_urls"] = {
            "en": f"/en/sup-rental-{loc.code}/",
            "es": f"/es/alquiler-sup-{loc.code}/",
        }
        # Етап 4: weather-стрічка + paddle-гайд + JSON-LD
        ctx["weather"] = WeatherSpot.objects.filter(location=loc).first()
        ctx["overview"] = GuideOverview.objects.filter(location=loc).first()
        ctx["routes"] = SupRoute.objects.filter(location=loc).prefetch_related("waypoints")
        ctx["beach_spots"] = BeachSpotIntel.objects.filter(location=loc)
        ctx["route_category"] = "all"
        ctx["json_ld"] = build_location_jsonld(loc, tariffs, get_language())
        return ctx


class LocationGuideTabView(TemplateView):
    """HTMX-паршали вкладок гайда (routes/beaches/overview/safety).

    routes + ?category= → повертає лише список маршрутів (ціль #routes-list);
    без category → повна вкладка з панеллю фільтрів.
    """

    def dispatch(self, request, *args, **kwargs):
        self.location = resolve_location(kwargs["prefix"], kwargs["city"])
        return super().dispatch(request, *args, **kwargs)

    def get_template_names(self):
        tab = self.kwargs["tab"]
        if tab == "routes" and self.request.GET.get("category"):
            return ["catalog/partials/_guide_routes_list.html"]
        return [f"catalog/partials/_guide_tab_{tab}.html"]

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        loc = self.location
        tab = self.kwargs["tab"]
        ctx["location"] = loc
        if tab == "routes":
            category = self.request.GET.get("category", "all")
            qs = SupRoute.objects.filter(location=loc).prefetch_related("waypoints")
            if category != "all":
                qs = qs.filter(category=category)
            ctx["routes"] = qs
            ctx["route_category"] = category
        elif tab == "beaches":
            ctx["beach_spots"] = BeachSpotIntel.objects.filter(location=loc)
        elif tab == "overview":
            ctx["overview"] = GuideOverview.objects.filter(location=loc).first()
        elif tab == "safety":
            ctx["wind_windows"] = WindWindow.objects.all()
            ctx["maritime_laws"] = MaritimeLaw.objects.all()
        return ctx
EOF

/* TUYYO.CLUB — інтерактив (прогресивне посилення, без залежностей) */
(function () {
  'use strict';
  /* 1. Hero-відео: fallback для повільних з'єднань (Network Information API) */
  var conn = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
  var isSlow = conn && (conn.saveData || ['slow-2g', '2g', '3g'].indexOf(conn.effectiveType) > -1);
  document.querySelectorAll('[data-hero-video]').forEach(function (wrap) {
    if (isSlow) { wrap.parentNode.removeChild(wrap); return; }
    var video = wrap.querySelector('video');
    if (video) {
      video.addEventListener('error', function () { wrap.parentNode.removeChild(wrap); }, true);
    }
  });
  /* 2. Галерея: активний стан фільтрів */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-gallery-filter]');
    if (!btn) return;
    document.querySelectorAll('[data-gallery-filter]').forEach(function (b) { b.classList.remove('is-active'); });
    btn.classList.add('is-active');
  });
  /* 3. Галерея: лайтбокс */
  var lb = document.getElementById('gallery-lightbox');
  if (lb) {
    var openBox = function (card) {
      var img = document.getElementById('lb-image');
      img.src = card.dataset.image;
      img.alt = card.dataset.title;
      document.getElementById('lb-title').textContent = card.dataset.title;
      document.getElementById('lb-desc').textContent = card.dataset.desc;
      document.getElementById('lb-spot').textContent = card.dataset.spot;
      document.getElementById('lb-location').textContent = card.dataset.location;
      document.getElementById('lb-book').href = card.dataset.bookUrl;
      document.getElementById('lb-book-label').textContent = card.dataset.bookLabel;
      lb.classList.remove('hidden');
      lb.classList.add('flex');
      document.body.style.overflow = 'hidden';
      document.getElementById('gallery-lightbox-close').focus();
    };
    var closeBox = function () {
      lb.classList.add('hidden');
      lb.classList.remove('flex');
      document.body.style.overflow = '';
    };
    document.addEventListener('click', function (e) {
      var card = e.target.closest('[data-gallery-card]');
      if (card) { openBox(card); return; }
      if (e.target.closest('#gallery-lightbox-close')) { closeBox(); return; }
      if (e.target === lb) closeBox();
    });
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' && !lb.classList.contains('hidden')) closeBox();
    });
  }
  /* 4. FAQ-акордеон */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-faq-toggle]');
    if (!btn) return;
    var panel = document.getElementById(btn.dataset.faqTarget);
    if (!panel) return;
    var expanded = btn.getAttribute('aria-expanded') === 'true';
    btn.setAttribute('aria-expanded', String(!expanded));
    panel.classList.toggle('hidden');
    var chevron = btn.querySelector('svg');
    if (chevron) chevron.classList.toggle('rotate-180');
  });
  /* 5. Плавний скрол до якоря при завантаженні */
  if (window.location.hash) {
    var el = document.getElementById(window.location.hash.slice(1));
    if (el) setTimeout(function () { el.scrollIntoView({ behavior: 'smooth' }); }, 150);
  }
  /* 6. Guide-вкладки: активний стан (контент завантажує HTMX) */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-guide-tab]');
    if (!btn) return;
    document.querySelectorAll('[data-guide-tab]').forEach(function (b) { b.classList.remove('is-active'); });
    btn.classList.add('is-active');
  });
  /* 7. Фільтри маршрутів: активний стан */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-route-filter]');
    if (!btn) return;
    document.querySelectorAll('[data-route-filter]').forEach(function (b) { b.classList.remove('is-active'); });
    btn.classList.add('is-active');
  });
  /* 8. Картка маршруту: розгорнути/згорнути waypoints */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-route-toggle]');
    if (!btn) return;
    var panel = document.getElementById(btn.dataset.routeTarget);
    if (!panel) return;
    var expanded = btn.getAttribute('aria-expanded') === 'true';
    btn.setAttribute('aria-expanded', String(!expanded));
    panel.classList.toggle('hidden');
    var openLabel = btn.querySelector('[data-route-open]');
    var closeLabel = btn.querySelector('[data-route-close]');
    if (openLabel && closeLabel) {
      openLabel.classList.toggle('hidden');
      closeLabel.classList.toggle('hidden');
    }
    var chevron = btn.querySelector('[data-route-chevron]');
    if (chevron) chevron.classList.toggle('rotate-180');
  });
})();
