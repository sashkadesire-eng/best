@dataclass(frozen=True)
class PriceQuote:
    boards_total: int
    days: int
    unit_price: Decimal
    subtotal: Decimal
    delivery_fee: Decimal
    total: Decimal
    deposit: Decimal


class PricingService:
    @staticmethod
    def quote(tariff_code, boards, delivery_method, rental_days=1):
        try:
            tariff = Tariff.objects.get(code=tariff_code, is_active=True)
        except Tariff.DoesNotExist:
            return None
        site = SiteSettings.get_solo()
        days = max(2, int(rental_days or 2)) if tariff.code == "multi" else 1
        unit = Decimal(tariff.price_per_board)
        boards_total = sum(int(q or 0) for q in boards.values())
        subtotal = unit * boards_total * days
        fee = Decimal(site.delivery_fee) if delivery_method == "delivery" else Decimal(0)
        return PriceQuote(
            boards_total=boards_total,
            days=days,
            unit_price=unit,
            subtotal=subtotal,
            delivery_fee=fee,
            total=subtotal + fee,
            deposit=Decimal(site.deposit_amount),
        )


# ── Calendar / slots ────────────────────────────────────────────────────────
class CalendarService:
    @staticmethod
    def month_grid(year, month):
        """Сітка тижнів (Пн–Нд) для місяця; None — порожня клітинка."""
        first = date(year, month, 1)
        lead = (first.weekday() + 6) % 7
        days = py_calendar.monthrange(year, month)[1]
        cells = [None] * lead + [date(year, month, d) for d in range(1, days + 1)]
        while len(cells) % 7:
            cells.append(None)
        return [cells[i : i + 7] for i in range(0, len(cells), 7)]

    @staticmethod
    def slot_error(d, t_str, boards_total=1):
        """None — слот вільний; 'past' | 'notice' | 'full'."""
        now = timezone.now()
        tz = timezone.get_current_timezone()
        hh, mm = map(int, t_str.split(":"))
        slot_dt = timezone.make_aware(datetime.combine(d, dtime(hour=hh, minute=mm)), tz)
        if slot_dt <= now:
            return "past"
        if slot_dt - now < MIN_NOTICE:
            return "notice"
        taken = (
            Booking.objects.filter(date=d, time=t_str)
            .exclude(status="cancelled")
            .aggregate(s=models.Sum("boards_total"))["s"]
            or 0
        )
        if taken + boards_total > MAX_BOARDS_PER_SLOT:
            return "full"
        return None


# ── Booking creation (idempotent) ───────────────────────────────────────────
class BookingService:
    @staticmethod
    @transaction.atomic
    def create_from_state(state):
        d = state["data"]
        existing = Booking.objects.filter(idempotency_key=state["token"]).first()
        if existing:
            return existing
        quote = PricingService.quote(
            d["tariff_code"], d.get("boards") or {}, d.get("delivery_method") or "delivery",
            d.get("rental_days") or 1,
        )
        tariff = Tariff.objects.get(code=d["tariff_code"])
        location = Location.objects.get(code=d["location_code"])
        booking = Booking.objects.create(
            reference=BookingService._unique_reference(),
            idempotency_key=state["token"],
            tariff=tariff,
            rental_days=quote.days,
            date=date.fromisoformat(d["date"]),
            time=dtime.fromisoformat(d["time"]),
            location=location,
            delivery_method=d["delivery_method"],
            delivery_type=d.get("delivery_type") or "",
            beach_spot=d.get("beach_spot") or "",
            address=d.get("address") or "",
            full_name=d["full_name"],
            phone=f"{d['phone_cc']} {d['phone']}".strip(),
            id_passport=d["id_passport"],
            email=d.get("email") or "",
            send_voucher=bool(d.get("send_voucher", True)),
            payment_method=d["payment_method"],
            boards_total=quote.boards_total,
            unit_price=quote.unit_price,
            subtotal=quote.subtotal,
            delivery_fee=quote.delivery_fee,
            total_price=quote.total,
            deposit_amount=quote.deposit,
            status="pending_payment" if d["payment_method"] == "bizum" else "confirmed",
        )
        for code, qty in (d.get("boards") or {}).items():
            if qty and int(qty) > 0:
                board = Board.objects.get(code=code)
                BookingItem.objects.create(
                    booking=booking, board=board, quantity=int(qty),
                    unit_price=quote.unit_price, line_total=quote.unit_price * int(qty),
                )
        return booking

    @staticmethod
    def _unique_reference():
        for _ in range(10):
            ref = f"TY-{random.randint(1000, 9999)}"
            if not Booking.objects.filter(reference=ref).exists():
                return ref
        return f"TY-{uuid.uuid4().hex[:6].upper()}"

class BizumService:
    @staticmethod
    def qr_svg(booking):
        import qrcode
        from qrcode.image.svg import SvgPathImage

        site = SiteSettings.get_solo()
        phone = site.bizum_phone.replace("+", "").replace(" ", "")
        payload = f"bizum://pay?phone={phone}&amount={booking.total_price}&concept=TUYYO {booking.reference}"
        img = qrcode.make(payload, image_factory=SvgPathImage)
        buf = BytesIO()
        img.save(buf)
        svg = buf.getvalue().decode("utf-8")
        svg = svg.replace("<?xml version='1.0' encoding='UTF-8'?>", "").strip()
        return mark_safe(svg)

def is_htmx(request):
    return request.headers.get("HX-Request") == "true"


def step_url(step):
    if step == "tariff":
        return reverse("booking")
    return reverse("booking_step", kwargs={"step": step})


class BookingWizardView(View):
    shell = "bookings/wizard.html"

    def dispatch(self, request, *args, **kwargs):
        self.state = get_state(request)
        return super().dispatch(request, *args, **kwargs)

    # ── GET ──
    def get(self, request, step="tariff", *args, **kwargs):
        self._apply_prefill(request)
        step = self._guard(step)
        d = self.state["data"]
        if step == "datetime" and not d.get("date"):
            d["date"] = str(timezone.localdate())
        g = request.GET
        if g.get("month"):
            d["_month"] = g["month"]
        if g.get("date"):
            d["date"] = g["date"]
            if not g.get("month"):
                d.pop("_month", None)
        self.state["step"] = step
        save_state(request, self.state)
        if is_htmx(request):
            return self._bundle(step)
        return render(request, self.shell, self._step_context(step))

    # ── POST ──
    def post(self, request, step="tariff", *args, **kwargs):
        form = self._form(step, request.POST)
        if not form.is_valid():
            return self._bundle(step, form=form)
        self._merge(step, form.cleaned_data)
        if step == "confirm":
            booking = BookingService.create_from_state(self.state)
            NotificationService.notify_booking(booking)
            clear_state(request)
            target = reverse("booking_success", kwargs={"reference": booking.reference})
            if is_htmx(request):
                resp = HttpResponse("")
                resp.headers["HX-Redirect"] = target
                return resp
            return HttpResponseRedirect(target)
        next_step = STEPS[STEPS.index(step) + 1]
        self.state["step"] = next_step
        save_state(request, self.state)
        resp = self._bundle(next_step)
        if is_htmx(request):
            resp.headers["HX-Push-Url"] = step_url(next_step)
        return resp

    # ── prefill з ?location=&tariff=&board= (CTA лендінгів) ──
    def _apply_prefill(self, request):
        g, d = request.GET, self.state["data"]
        changed = False
        if g.get("tariff") and Tariff.objects.filter(code=g["tariff"], is_active=True).exists():
            d["tariff_code"], changed = g["tariff"], True
        if g.get("board") and Board.objects.filter(code=g["board"], is_active=True).exists():
            d.setdefault("boards", {})[g["board"]] = max(1, d["boards"].get(g["board"], 0))
            changed = True
        if g.get("location") and Location.objects.filter(code=g["location"], is_active=True).exists():
            if d.get("location_code") != g["location"]:
                d["beach_spot"] = ""
            d["location_code"], changed = g["location"], True
        if g.get("delivery_method") in ("delivery", "pickup"):
            if d.get("delivery_method") != g["delivery_method"]:
                d["delivery_method"] = g["delivery_method"]
                if g["delivery_method"] == "delivery" and not d.get("delivery_type"):
                    d["delivery_type"] = "beach"
                changed = True
        if g.get("delivery_type") in ("beach", "hotel"):
            d["delivery_type"], changed = g["delivery_type"], True
        if changed:
            save_state(request, self.state)

    # ── guard: не далі першого незаповненого кроку ──
    def _guard(self, step):
        if step not in STEPS:
            step = "tariff"
        first = self._first_incomplete()
        if STEPS.index(step) > STEPS.index(first):
            return first
        return step

    def _first_incomplete(self):
        d = self.state["data"]
        if not d.get("tariff_code"):
            return "tariff"
        if not any((d.get("boards") or {}).values()):
            return "boards"
        if not d.get("date") or not d.get("time"):
            return "datetime"
        if not d.get("location_code"):
            return "delivery"
        if d.get("delivery_method") == "delivery":
            if d.get("delivery_type") == "hotel" and not d.get("address"):
                return "delivery"
            if d.get("delivery_type") != "hotel" and not d.get("beach_spot"):
                return "delivery"
        if not d.get("full_name") or not d.get("phone") or not d.get("id_passport"):
            return "customer"
        return "confirm"

    # ── форми ──
    def _form(self, step, data):
        d = self.state["data"]
        if step == "tariff":
            return TariffStepForm(data)
        if step == "boards":
            return BoardsStepForm(data)
        if step == "datetime":
            boards_total = sum((d.get("boards") or {}).values()) or 1
            return DateTimeStepForm(data, boards_total=boards_total)
        if step == "delivery":
            return DeliveryStepForm(data)
        if step == "customer":
            return CustomerStepForm(data)
        return ConfirmStepForm(data)

    def _merge(self, step, c):
        d = self.state["data"]
        if step == "tariff":
            d["tariff_code"] = c["tariff"]
            d["rental_days"] = c.get("rental_days") or 2 if c["tariff"] == "multi" else 1
        elif step == "boards":
            d["boards"] = {k[4:]: int(v or 0) for k, v in c.items() if k.startswith("qty_")}
        elif step == "datetime":
            d["date"], d["time"] = str(c["date"]), c["time"]
        elif step == "delivery":
            d["location_code"] = c["location"]
            d["delivery_method"] = c["delivery_method"]
            d["delivery_type"] = c.get("delivery_type") or ""
            d["beach_spot"] = c.get("beach_spot") or ""
            d["address"] = c.get("address") or ""
        elif step == "customer":
            d.update(
                full_name=c["full_name"], phone_cc=c["phone_cc"], phone=c["phone"],
                id_passport=c["id_passport"], email=c.get("email") or "",
                send_voucher=bool(c.get("send_voucher")),
            )
        elif step == "confirm":
            d["payment_method"] = c["payment_method"]

    # ── контексти ──
    def _quote(self):
        d = self.state["data"]
        if not d.get("tariff_code") or not any((d.get("boards") or {}).values()):
            return None
        return PricingService.quote(
            d["tariff_code"], d.get("boards") or {},
            d.get("delivery_method") or "delivery", d.get("rental_days") or 1,
        )

    def _summary_context(self):
        d = self.state["data"]
        tariff = Tariff.objects.filter(code=d.get("tariff_code")).first()
        location = Location.objects.filter(code=d.get("location_code")).first()
        boards = []
        for code, qty in (d.get("boards") or {}).items():
            if qty and int(qty) > 0:
                b = Board.objects.filter(code=code).first()
                if b:
                    boards.append({"name": b.name, "qty": int(qty)})
        site = SiteSettings.get_solo()
        return {
            "quote": self._quote(),
            "tariff_name": tariff.name if tariff else "—",
            "is_multi": d.get("tariff_code") == "multi",
            "rental_days": d.get("rental_days") or 1,
            "boards_sel": boards,
            "date": d.get("date") or "",
            "time": d.get("time") or "",
            "location_name": location.name if location else "—",
            "delivery_method": d.get("delivery_method"),
            "delivery_detail": d.get("beach_spot") or d.get("address") or "",
            "deposit": site.deposit_amount,
        }

    def _steps_meta(self, step):
        first = self._first_incomplete()
        meta = []
        for code, label, ic in STEP_META:
            if code == step:
                st = "current"
            elif STEPS.index(code) < STEPS.index(first):
                st = "done"
            else:
                st = "todo"
            meta.append({"code": code, "label": label, "icon": ic, "state": st, "url": step_url(code)})
        return meta

    def _calendar_context(self):
        from datetime import date as date_cls, timedelta

        d = self.state["data"]
        today = timezone.localdate()
        try:
            selected = date_cls.fromisoformat(d["date"]) if d.get("date") else today
        except ValueError:
            selected = today
        try:
            vy, vm = map(int, (d.get("_month") or "").split("-"))
        except ValueError:
            vy, vm = selected.year, selected.month
        weeks = CalendarService.month_grid(vy, vm)
        boards_total = sum((d.get("boards") or {}).values()) or 1
        groups = []
        for name, times in SLOT_GROUPS:
            slots = []
            for t in times:
                err = CalendarService.slot_error(selected, t, boards_total)
                slots.append({"time": t, "disabled": err is not None, "reason": err})
            groups.append({"name": name, "slots": slots})
        prev_m = date_cls(vy, vm, 1) - timedelta(days=1)
        next_m = date_cls(vy, vm, 28) + timedelta(days=7)
        return {
            "weeks": weeks,
            "selected_date": selected,
            "today": today,
            "view_date": date_cls(vy, vm, 1),
            "view_month_label": f"{vy}-{vm:02d}",
            "prev_month": f"{prev_m.year}-{prev_m.month:02d}",
            "next_month": f"{next_m.year}-{next_m.month:02d}",
            "slot_groups": groups,
            "checked_time": d.get("time") or "",
        }

    def _step_context(self, step, form=None):
        d = self.state["data"]
        bound = form
        form = form or self._form(step, None)
        ctx = {
            "step": step,
            "step_template": f"bookings/partials/_step_{step}.html",
            "step_index": STEPS.index(step) + 1,
            "steps_meta": self._steps_meta(step),
            "form": form,
            "data": d,
            "summary": self._summary_context(),
            "site_settings": SiteSettings.get_solo(),
            "post_url": step_url(step),
            "self_get_url": step_url(step),
            "back_url": step_url(STEPS[STEPS.index(step) - 1]) if step != "tariff" else None,
        }
        if step == "tariff":
            ctx["tariffs"] = Tariff.objects.filter(is_active=True)
        elif step == "boards":
            ctx["boards_list"] = Board.objects.filter(is_active=True)
        elif step == "datetime":
            ctx.update(self._calendar_context())
            if bound is not None and bound.is_bound:
                ctx["checked_time"] = bound["time"].value() or d.get("time") or ""
        elif step == "delivery":
            ctx["locations"] = Location.objects.filter(is_active=True).prefetch_related("beaches")
            ctx["current_location"] = Location.objects.filter(code=d.get("location_code")).first()
        return ctx

    def _bundle(self, step, form=None):
        return render(self.request, "bookings/partials/_step_bundle.html", self._step_context(step, form))
