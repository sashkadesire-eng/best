```

chmod +x stage8_content_hub.sh && ./stage8_content_hub.sh
```

class ArticleCategory(TranslatableModel):
    code = models.SlugField(unique=True)
    order = models.PositiveIntegerField(default=0)
    translations = TranslatedFields(
        name=models.CharField(max_length=120),
        description=models.CharField(max_length=250, blank=True),

        ordering = ["order"]
        verbose_name_plural = "article categories"

"""Галерея-фільтр (Етап 3) + контент-хаб (Етап 8)."""
from django.http import Http404
from django.utils import timezone
from django.utils.translation import get_language
from django.views.generic import TemplateView

from content.models import Article, ArticleCategory, GalleryItem


class GalleryGridView(TemplateView):
    """HTMX partial: фільтр галереї (логіка 1:1 з GallerySection.tsx)."""
    template_name = "content/partials/_gallery_grid.html"
    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        category = self.request.GET.get("category", "all")
        qs = GalleryItem.objects.filter(is_active=True).select_related("location")
        if category != "all":
            from django.db.models import Q
            qs = qs.filter(Q(category=category) | Q(location__code=category))
        ctx["gallery_items"] = qs
        ctx["active_category"] = category
        return ctx


def _published_articles():
    return (
        Article.objects.filter(is_active=True, published_at__lte=timezone.localdate())
        .select_related("category")
        .prefetch_related("locations")
    )


class ArticleListView(TemplateView):
    template_name = "content/blog_list.html"

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        category = self.request.GET.get("category", "all")
        qs = _published_articles()
        featured = qs.filter(is_featured=True).first()
        if category != "all":
            qs = qs.filter(category__code=category)
        grid = qs.order_by("-published_at")
        if featured and category == "all":
            grid = grid.exclude(pk=featured.pk)
        categories = []
        for cat in ArticleCategory.objects.all():
            categories.append({
                "code": cat.code,
                "name": cat.name,
                "count": _published_articles().filter(category=cat).count(),
            })
        ctx.update({
            "featured": featured,
            "articles": grid,
            "categories": categories,
            "active_category": category,
            "total_articles": _published_articles().count(),
            "lang_urls": {"en": "/en/blog/", "es": "/es/blog/"},
            "meta_title": "The Field Journal — Costa Blanca SUP Guides | TUYYO.CLUB",
            "meta_description": "Route beta, gear truth and wind reading from local Costa Blanca paddlers. Benidorm, Altea, Villajoyosa, Calpe & Alicante.",
        })
        return ctx

    def get(self, request, *args, **kwargs):
        # HTMX → повертаємо лише сітку
        if request.headers.get("HX-Request") == "true":
            self.template_name = "content/partials/_articles_grid.html"
        return super().get(request, *args, **kwargs)


class ArticleDetailView(TemplateView):
    template_name = "content/blog_detail.html"

    def dispatch(self, request, *args, **kwargs):
        lang = get_language()
        self.article = (
            Article.objects.filter(
                is_active=True,
                translations__language_code=lang,
                translations__slug=kwargs["slug"],
            )
            .select_related("category")
            .prefetch_related("locations")
            .distinct()
            .first()
        )
        if self.article is None:
            raise Http404("Статтю не знайдено для цієї мови")
        return super().dispatch(request, *args, **kwargs)

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        a = self.article
        related = (
            _published_articles()
            .filter(category=a.category)
            .exclude(pk=a.pk)
            .order_by("-published_at")[:3]
        )
        site_url = "https://tuyyo.club"
        json_ld = {
            "@context": "https://schema.org",
            "@graph": [
                {
                    "@type": "BreadcrumbList",
                    "itemListElement": [
                        {"@type": "ListItem", "position": 1, "name": "Home", "item": f"{site_url}/{get_language()}/"},
                        {"@type": "ListItem", "position": 2, "name": "Journal", "item": f"{site_url}/{get_language()}/blog/"},
                        {"@type": "ListItem", "position": 3, "name": a.title},
                    ],
                },
                {
                    "@type": "Article",
                    "headline": a.title,
                    "image": a.image_url,
                    "datePublished": a.published_at.isoformat(),
                    "author": {"@type": "Organization", "name": "TUYYO.CLUB"},
                    "publisher": {"@type": "Organization", "name": "TUYYO GROUP S.L."},
                    "description": a.excerpt,
                    "mainEntityOfPage": f"{site_url}{a.get_absolute_url()}",
                },
            ],
        }
        import json as _json
        ctx.update({
            "article": a,
            "related": related,
            "json_ld": _json.dumps(json_ld, ensure_ascii=False).replace("</", "<\\/"),
            "meta_title": a.meta_title or a.title,
            "meta_description": a.meta_description or a.excerpt,
            "lang_urls": {
                tr.language_code: f"/{tr.language_code}/blog/{tr.slug}/"
                for tr in a.translations.all() if tr.slug
            },
        })
        return ctx
EOF

from django.contrib.sitemaps import Sitemap
from django.utils.translation import get_language

from content.models import Article


class ArticleSitemap(Sitemap):
    i18n = True
    protocol = "https"
    changefreq = "weekly"
    priority = 0.7

    def items(self):
        return Article.objects.filter(is_active=True).order_by("-published_at")

    def location(self, obj):
        return f"/{get_language()}/blog/{obj.slug}/"

    def lastmod(self, obj):
        return obj.published_at

class Command(BaseCommand):
    help = "Seed контент-хабу: категорії + 6 статей EN/ES"

    def handle(self, *args, **options):
        # Категорії
        cat_objs = {}
        for code, en, es, order in CATEGORIES:
            cat, _ = ArticleCategory.objects.get_or_create(code=code, defaults={"order": order})
            cat.set_current_language("en"); cat.name = en
            cat.set_current_language("es"); cat.name = es
            cat.save()
            cat_objs[code] = cat

        # Пункт меню «Journal»
        link, _ = MenuLink.objects.get_or_create(
            url="/blog/", placement="navbar", defaults={"order": 45, "icon": "compass"}
        )
        link.set_current_language("en"); link.label = "Journal"
        link.set_current_language("es"); link.label = "Diario"
        link.save()

        # Статті
        for a in ARTICLES:
            en_slug = a["en"]["slug"]
            existing = Article.objects.filter(
                translations__language_code="en", translations__slug=en_slug
            ).first()
            if existing:
                continue
            art = Article.objects.create(
                category=cat_objs[a["cat"]],
                author=a["author"],
                published_at=a["date"],
                read_minutes=a["read"],
                image_url=a["img"],
                is_featured=a["featured"],
                is_active=True,
            )
            for lang in ("en", "es"):
                d = a[lang]
                art.set_current_language(lang)
                art.slug = d["slug"]
                art.title = d["title"]
                art.excerpt = d["excerpt"]
                art.body = d["body"]
                art.meta_title = d["meta_title"]
                art.meta_description = d["meta_description"]
            art.save()
            for loc_code in a["locs"]:
                loc = Location.objects.filter(code=loc_code).first()
                if loc:
                    art.locations.add(loc)

        self.stdout.write(self.style.SUCCESS(f"✅ Articles seeded: {Article.objects.count()}"))
