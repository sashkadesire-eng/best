"""Жива погода Open-Meteo (без API-ключа, без cookies — GDPR-friendly).

Стратегія: API → запис у WeatherSpot → усі шаблони (hero-стрічка, оверлеї
локацій) автоматично показують живі дані. Помилка мережі → зберігаються
попередні значення (fail-safe, сайт ніколи не падає через погоду).
"""
import json
import urllib.request

FORECAST_URL = "https://api.open-meteo.com/v1/forecast"
MARINE_URL = "https://marine-api.open-meteo.com/v1/marine"

COMPASS = ["N", "NE", "E", "SE", "S", "SW", "W", "NW"]


def degrees_to_compass(degrees):
    return COMPASS[int((degrees + 22.5) % 360 // 45)]


def compute_status(wind_knots, wave_height):
    """perfect / good / caution — узгоджено з гарантією 15 вузлів."""
    if wind_knots >= 15 or (wave_height is not None and wave_height >= 1.5):
        return "caution"
    if wind_knots >= 9 or (wave_height is not None and wave_height >= 0.5):
        return "good"
    return "perfect"


def _get_json(url, timeout=8):
    req = urllib.request.Request(url, headers={"User-Agent": "TUYYO.CLUB/1.0"})
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        return json.loads(resp.read().decode("utf-8"))


def fetch_spot(lat, lng):
    """Прогноз + морські дані для координат. None — якщо мережа недоступна."""
    try:
        forecast = _get_json(
            f"{FORECAST_URL}?latitude={lat}&longitude={lng}"
            "&current=temperature_2m,wind_speed_10m,wind_direction_10m"
        )
        current = forecast.get("current") or {}
        if "temperature_2m" not in current:
            return None
        marine_current = {}
        try:
            marine = _get_json(
                f"{MARINE_URL}?latitude={lat}&longitude={lng}"
                "&current=wave_height,sea_surface_temperature"
            )
            marine_current = marine.get("current") or {}
        except Exception:
            marine_current = {}
        wind_knots = round((current.get("wind_speed_10m") or 0) / 1.852)
        wave = marine_current.get("wave_height")
        sst = marine_current.get("sea_surface_temperature")
        return {
            "temp": round(current["temperature_2m"]),
            "water_temp": round(sst) if sst is not None else None,
            "wind_knots": wind_knots,
            "wind_direction": degrees_to_compass(current.get("wind_direction_10m") or 0),
            "wave_height": f"{wave:.1f}m" if wave is not None else None,
            "status": compute_status(wind_knots, wave),
        }
    except Exception:
        return None


def update_all_spots():
    """Оновлює всі WeatherSpot з API. Повертає (updated, failed)."""
    from content.models import WeatherSpot

    updated, failed = 0, 0
    for spot in WeatherSpot.objects.select_related("location"):
        loc = spot.location
        if loc.lat is None or loc.lng is None:
            failed += 1
            continue
        data = fetch_spot(loc.lat, loc.lng)
        if not data:
            failed += 1
            continue
        spot.temp = data["temp"]
        if data["water_temp"] is not None:
            spot.water_temp = data["water_temp"]
        spot.wind_knots = data["wind_knots"]
        spot.wind_direction = data["wind_direction"]
        if data["wave_height"]:
            spot.wave_height = data["wave_height"]
        spot.status = data["status"]
        spot.source = "open-meteo"
        spot.save()
        updated += 1
    return updated, failed
EOF

import math
import os

from PIL import Image, ImageDraw

os.makedirs("static/icons", exist_ok=True)


def base(size):
    img = Image.new("RGBA", (size, size), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    top, bot = (2, 132, 199), (13, 148, 136)  # sky-600 → teal-600
    for y in range(size):
        t = y / max(1, size - 1)
        col = tuple(int(top[i] + (bot[i] - top[i]) * t) for i in range(3)) + (255,)
        d.line([(0, y), (size - 1, y)], fill=col)
    return img, d


def draw_waves(d, size):
    specs = [(0.40, 255, 0.085), (0.54, 215, 0.065), (0.68, 175, 0.05)]
    for k, (yc, alpha, w) in enumerate(specs):
        pts = []
        amp = size * 0.05
        step = max(1, size // 120)
        for x in range(0, size + 1, step):
            y = size * yc + math.sin((x / size) * 2 * math.pi + k * 1.3) * amp
            pts.append((x, y))
        d.line(pts, fill=(255, 255, 255, alpha), width=max(2, int(size * w)))


def rounded(img, radius):
    mask = Image.new("L", img.size, 0)
    ImageDraw.Draw(mask).rounded_rectangle(
        [0, 0, img.size[0] - 1, img.size[1] - 1], radius, fill=255
    )
    out = img.copy()
    out.putalpha(mask)
    return out


for size, name, maskable in [
    (192, "icon-192.png", False),
    (512, "icon-512.png", False),
    (512, "icon-maskable-512.png", True),
    (180, "apple-touch-icon.png", False),
]:
    img, d = base(size)
    draw_waves(d, size)
    if not maskable:
        img = rounded(img, int(size * 0.22))
    img.save(f"static/icons/{name}")
print("✅ PWA-іконки: 192 / 512 / maskable / apple-touch")

/* TUYYO.CLUB Service Worker — offline-first PWA (Етап 7) */
const CACHE = 'tuyyo-v1';
const OFFLINE_URL = '/offline/';
const PRECACHE = [
  OFFLINE_URL,
  '/en/',
  '/static/css/app.css',
  '/static/js/tuyyo.js',
  '/static/js/htmx.min.js',
  '/static/manifest.webmanifest',
  '/static/icons/icon-192.png',
  '/static/icons/icon-512.png'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE)
      .then((cache) => cache.addAll(PRECACHE))
      .then(() => self.skipWaiting())
  );
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys()
      .then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k))))
      .then(() => self.clients.claim())
  );
});

self.addEventListener('message', (event) => {
  if (event.data === 'SKIP_WAITING') self.skipWaiting();
});

self.addEventListener('fetch', (event) => {
  const req = event.request;
  if (req.method !== 'GET') return;
  const url = new URL(req.url);
  if (url.origin !== self.location.origin) return; /* зовнішні (Plausible, шрифти) — повз кеш */

  /* Сторінки: network-first + кеш → offline-сторінка */
  const accept = req.headers.get('accept') || '';
  if (accept.includes('text/html')) {
    event.respondWith(
      fetch(req)
        .then((res) => {
          const copy = res.clone();
          caches.open(CACHE).then((cache) => cache.put(req, copy));
          return res;
        })
        .catch(() => caches.match(req).then((hit) => hit || caches.match(OFFLINE_URL)))
    );
    return;
  }

  /* Статика: cache-first з довантаженням у кеш */
  event.respondWith(
    caches.match(req).then((hit) => {
      if (hit) return hit;
      return fetch(req).then((res) => {
        if (res.ok) {
          const copy = res.clone();
          caches.open(CACHE).then((cache) => cache.put(req, copy));
        }
        return res;
      });
    })
  );
});
