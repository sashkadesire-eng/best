# Етап 9 — Продакшн-деплой: скрипт повного відтворення

Запуск (з `~/projects`):

```
nano stage9_production.sh        # вставте скрипт, збережіть
chmod +x stage9_production.sh && ./stage9_production.sh
```

Скрипт: багатоступеневий **Dockerfile** (Tailwind-збірка + compilemessages + collectstatic з WhiteNoise-manifestом → компактний runtime без build-залежностей) → **docker-compose** з Postgres 16 і healthcheck'ами → посилений **prod.py** (HSTS, CSRF-trusted origins, Redis-кеш за наявності, SMTP, Sentry з Django-інтеграцією, структуровані логи) → **entrypoint.sh** (міграції + ідемпотентний seed + gunicorn) → ендпоінт **`/healthz/`** зі статусом БД → **Lighthouse-бюджети** + CI з docker-build → **nginx/Caddy** конфіги з автосертифікатами → вичерпний **DEPLOY.md** і чеклист env-змінних → ADR-0008.

"""Продакшн-налаштування: Docker + gunicorn + WhiteNoise + Postgres + Sentry.

Усі секрети — через env. За замовчуванням передбачається робота за
reverse-proxy (nginx/Caddy), який термінує TLS.

import environ

from .base import *  # noqa: F403

DEBUG = False

SESSION_COOKIE_SECURE = True
CSRF_COOKIE_SECURE = True
SECURE_HSTS_SECONDS = 31536000
SECURE_HSTS_INCLUDE_SUBDOMAINS = True
SECURE_HSTS_PRELOAD = True
X_FRAME_OPTIONS = "DENY"
SECURE_CONTENT_TYPE_NOSNIFF = True
SECURE_REFERRER_POLICY = "strict-origin-when-cross-origin"

# ── Статика через WhiteNoise (стиснення + річний кеш) ──────────────────────
STORAGES = {
    "default": {"BACKEND": "django.core.files.storage.FileSystemStorage"},
    "staticfiles": {"BACKEND": "whitenoise.storage.CompressedManifestStaticFilesStorage"},
}
WHITENOISE_MAX_AGE = 31536000

# ── Кеш: Redis за наявності REDIS_URL, інакше LocMem ───────────────────────
REDIS_URL = env("REDIS_URL", default="")
if REDIS_URL:
    CACHES = {
        "default": {
            "BACKEND": "django_redis.cache.RedisCache",
            "LOCATION": REDIS_URL,
            "OPTIONS": {"CLIENT_CLASS": "django_redis.client.DefaultClient"},
        }
    }
else:
    CACHES = {"default": {"BACKEND": "django.core.cache.backends.locmem.LocMemCache"}}

# ── Email: SMTP у продакшні ────────────────────────────────────────────────
EMAIL_BACKEND = env("EMAIL_BACKEND", default="django.core.mail.backends.smtp.EmailBackend")
EMAIL_HOST = env("EMAIL_HOST", default="")
EMAIL_PORT = env.int("EMAIL_PORT", default=587)
EMAIL_HOST_USER = env("EMAIL_HOST_USER", default="")
EMAIL_HOST_PASSWORD = env("EMAIL_HOST_PASSWORD", default="")
EMAIL_USE_TLS = env.bool("EMAIL_USE_TLS", default=True)
DEFAULT_FROM_EMAIL = env("DEFAULT_FROM_EMAIL", default="TUYYO.CLUB <tuyyogroup@gmail.com>")

# ── Sentry (опційно) ───────────────────────────────────────────────────────
SENTRY_DSN = env("SENTRY_DSN", default="")
if SENTRY_DSN:
    import sentry_sdk
    from sentry_sdk.integrations.django import DjangoIntegration

    sentry_sdk.init(
        dsn=SENTRY_DSN,
        integrations=[DjangoIntegration()],
        traces_sample_rate=0.1,
        send_default_pii=False,
        environment="production",
    )

# ── Структуровані логи ─────────────────────────────────────────────────────
LOGGING = {
    "version": 1,
    "disable_existing_loggers": False,
    "formatters": {
        "verbose": {"format": "{levelname} {asctime} {module} {message}", "style": "{"},
    },
    "handlers": {"console": {"class": "logging.StreamHandler", "formatter": "verbose"}},
    "root": {"handlers": ["console"], "level": "INFO"},
    "loggers": {

    },

# ── 03. /healthz/ — ендпоінт здоров'я ──────────────────────────────────────
python - << 'PY'
import pathlib

p = pathlib.Path("core/views.py")
text = p.read_text(encoding="utf-8")
if "def healthz" not in text:
    text += '''

def healthz(request):
    """Ендпоінт здоров'я для Docker/orchestrator: статус + перевірка БД."""
    from django.db import connection

    db_ok = True
    try:
        with connection.cursor() as cursor:
            cursor.execute("SELECT 1")
    except Exception:
        db_ok = False
    from django.http import JsonResponse

    return JsonResponse(
        {
            "status": "ok" if db_ok else "degraded",
            "database": "up" if db_ok else "down",
            "version": "1.0.0",
        },
        status=200 if db_ok else 503,
    )
'''
    p.write_text(text, encoding="utf-8")
    print("✅ core/views.py: додано healthz")

p = pathlib.Path("config/urls.py")
text = p.read_text(encoding="utf-8")
if "healthz" not in text:
    text = text.replace(
        "from core.views import OfflineView, StyleguideView",
        "from core.views import OfflineView, StyleguideView, healthz",
    )
    text = text.replace(
        '    path("robots.txt", TemplateView.as_view(template_name="robots.txt", content_type="text/plain")),',
        '    path("robots.txt", TemplateView.as_view(template_name="robots.txt", content_type="text/plain")),\n'
        '    path("healthz/", healthz, name="healthz"),',
    )
    p.write_text(text, encoding="utf-8")
    print("✅ config/urls.py: підключено /healthz/")
PY

# ── 04. Dockerfile (multi-stage) ───────────────────────────────────────────
cat > Dockerfile << 'EOF'
# syntax=docker/dockerfile:1
# ═══════════ Stage 1: build (Tailwind, i18n, collectstatic) ═══════════
FROM python:3.12-slim AS build
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    DJANGO_SETTINGS_MODULE=config.settings.dev
WORKDIR /app
RUN apt-get update && apt-get install -y --no-install-recommends \
        curl gettext \
    && rm -rf /var/lib/apt/lists/*
COPY requirements.txt requirements-prod.txt ./
RUN pip install --no-cache-dir -r requirements-prod.txt
COPY . .
# Збірка Tailwind CSS standalone-компілятором
RUN curl -sfL -o /tmp/tailwindcss \
        https://github.com/tailwindlabs/tailwindcss/releases/latest/download/tailwindcss-linux-x64 \
    && chmod +x /tmp/tailwindcss \
    && /tmp/tailwindcss -i static/css/input.css -o static/css/app.css --minify
# Компіляція перекладів (.po -> .mo)
RUN python manage.py compilemessages
# Collectstatic з WhiteNoise-manifestом (потрібні prod-налаштування)
RUN DJANGO_SETTINGS_MODULE=config.settings.prod \
    SECRET_KEY=build-time-secret \
    DATABASE_URL=sqlite:///build.sqlite3 \
    python manage.py collectstatic --noinput
RUN chmod +x entrypoint.sh

# ═══════════ Stage 2: runtime (без build-залежностей) ═══════════
FROM python:3.12-slim AS runtime
ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    DJANGO_SETTINGS_MODULE=config.settings.prod \
    PORT=8000
WORKDIR /app
RUN apt-get update && apt-get install -y --no-install-recommends \
        libpq5 gettext \
    && rm -rf /var/lib/apt/lists/* \
    && addgroup --system app && adduser --system --ingroup app app
COPY --from=build /usr/local/lib/python3.12/site-packages /usr/local/lib/python3.12/site-packages
COPY --from=build /usr/local/bin /usr/local/bin
COPY --from=build --chown=app:app /app /app
USER app
EXPOSE 8000
HEALTHCHECK --interval=30s --timeout=5s --start-period=20s --retries=3 \
    CMD python -c "import urllib.request,sys; sys.exit(0 if urllib.request.urlopen('http://localhost:8000/healthz/', timeout=3).status==200 else 1)"
CMD ["./entrypoint.sh"]
EOF

#!/usr/bin/env bash

echo "==> Сід CMS-контенту (ідемпотентно)…"
python manage.py seed_cms
python manage.py seed_content
python manage.py seed_home
python manage.py seed_articles || true

echo "==> Запуск gunicorn…"

    --bind "0.0.0.0:${PORT:-8000}" \
    --workers "${WEB_CONCURRENCY:-3}" \
    --threads 2 \
    --timeout 60 \
    --graceful-timeout 30 \
    --access-logfile - \
    --error-logfile -
EOF

cat > deploy/nginx.conf << 'EOF'
# TUYYO.CLUB — nginx reverse-proxy (TLS-термінація → gunicorn/WhiteNoise)
upstream tuyyo_web {
    server web:8000;
}

server {
    listen 80;
    server_name tuyyo.club www.tuyyo.club;
    return 301 https://tuyyo.club$request_uri;
}

server {
    listen 443 ssl http2;
    server_name tuyyo.club;

    ssl_certificate     /etc/letsencrypt/live/tuyyo.club/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/tuyyo.club/privkey.pem;
    ssl_protocols       TLSv1.2 TLSv1.3;

    client_max_body_size 20m;
    gzip on;
    gzip_types text/css application/javascript application/json image/svg+xml;

    location /static/ {
        # WhiteNoise вже віддає статику; можна прискорити через proxy_cache
        proxy_pass http://tuyyo_web;
        proxy_set_header Host $host;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    location / {
        proxy_pass http://tuyyo_web;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_read_timeout 60s;
    }
}
EOF

# TUYYO.CLUB — Caddy (автоматичні Let's Encrypt сертифікати)
