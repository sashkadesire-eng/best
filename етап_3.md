"""PageView-рушій + HomeView із глобальним контентом для DB-секцій."""
from django.conf import settings
from django.http import Http404
from django.utils.translation import get_language
from django.views.generic import TemplateView

from catalog.models import Board, Location, Tariff
from cms.models import Page, SiteSettings
from content.models import FAQItem, GalleryItem, Review, WeatherSpot


def active_sections(page):
    return page.sections.filter(is_active=True).order_by("order").prefetch_related("items")


def page_lang_urls(page):
    """URL сторінки кожною мовою (локалізовані slug'и, ADR-0007)."""
    if page.is_homepage:
        return {code: f"/{code}/" for code, _ in settings.LANGUAGES}
    urls = {}
    for tr in page.translations.all():
        if tr.slug:
            urls[tr.language_code] = f"/{tr.language_code}/{tr.slug}/"
    return urls


def page_context(page):
    ctx = {"page": page, "sections": active_sections(page), "lang_urls": page_lang_urls(page)}
    if page.meta_title:
        ctx["meta_title"] = page.meta_title
    if page.meta_description:
        ctx["meta_description"] = page.meta_description
    return ctx


class HomeView(TemplateView):
    template_name = "cms/page_home.html"

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        page = Page.objects.filter(is_homepage=True, published=True).first()
        if page:
            ctx.update(page_context(page))
        else:
            ctx["lang_urls"] = {code: f"/{code}/" for code, _ in settings.LANGUAGES}
            ctx["sections"] = []
        # Глобальний контент для секцій (тарифи/флот/галерея/локації+погода/відгуки/FAQ)
        ctx["tariffs"] = Tariff.objects.filter(is_active=True)
        ctx["boards"] = Board.objects.filter(is_active=True)
        ctx["faqs"] = FAQItem.objects.filter(is_active=True)
        ctx["reviews"] = Review.objects.filter(is_active=True)
        ctx["gallery_items"] = GalleryItem.objects.filter(is_active=True).select_related("location")
        ctx["active_category"] = "all"
        weather = {w.location_id: w for w in WeatherSpot.objects.all()}
        locations = Location.objects.filter(is_active=True).prefetch_related("beaches")
        ctx["locations_with_weather"] = [(loc, weather.get(loc.id)) for loc in locations]
        return ctx


class PageDetailView(TemplateView):
    """Суворий мовний резолвер: /es/privacy-policy/ → 404 (ES-slug інший)."""

    def dispatch(self, request, *args, **kwargs):
        lang = get_language()
        self.page = (
            Page.objects.filter(
                published=True,
                is_homepage=False,
                translations__language_code=lang,
                translations__slug=kwargs["slug"],
            )
            .distinct()
            .first()
        )
        if self.page is None:
            raise Http404("Сторінку не знайдено для цієї мови")
        return super().dispatch(request, *args, **kwargs)

    def get_template_names(self):
        return [self.page.template]

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        ctx.update(page_context(self.page))
        return ctx
EOF

from django.db.models import Q
from django.views.generic import TemplateView

from content.models import GalleryItem


class GalleryGridView(TemplateView):
    """HTMX partial: фільтр галереї (логіка 1:1 з GallerySection.tsx)."""

    template_name = "content/partials/_gallery_grid.html"

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        category = self.request.GET.get("category", "all")
        qs = GalleryItem.objects.filter(is_active=True).select_related("location")
        if category != "all":
            qs = qs.filter(Q(category=category) | Q(location__code=category))
        ctx["gallery_items"] = qs
        ctx["active_category"] = category
        return ctx

/* TUYYO.CLUB — інтерактив секцій (прогресивне посилення, без залежностей) */
(function () {
  'use strict';

  /* 1. Hero-відео: fallback для повільних з'єднань (Network Information API, 1:1 HomePage.tsx) */
  var conn = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
  var isSlow = conn && (conn.saveData || ['slow-2g', '2g', '3g'].indexOf(conn.effectiveType) > -1);
  document.querySelectorAll('[data-hero-video]').forEach(function (wrap) {
    if (isSlow) { wrap.parentNode.removeChild(wrap); return; }
    var video = wrap.querySelector('video');
    if (video) {
      video.addEventListener('error', function () { wrap.parentNode.removeChild(wrap); }, true);
    }
  });

  /* 2. Галерея: активний стан фільтрів (сітку завантажує HTMX) */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-gallery-filter]');
    if (!btn) return;
    document.querySelectorAll('[data-gallery-filter]').forEach(function (b) { b.classList.remove('is-active'); });
    btn.classList.add('is-active');
  });

  /* 3. Галерея: лайтбокс (делегування — картки можуть бути замінені HTMX) */
  var lb = document.getElementById('gallery-lightbox');
  if (lb) {
    var openBox = function (card) {
      var img = document.getElementById('lb-image');
      img.src = card.dataset.image;
      img.alt = card.dataset.title;
      document.getElementById('lb-title').textContent = card.dataset.title;
      document.getElementById('lb-desc').textContent = card.dataset.desc;
      document.getElementById('lb-spot').textContent = card.dataset.spot;
      document.getElementById('lb-location').textContent = card.dataset.location;
      document.getElementById('lb-book').href = card.dataset.bookUrl;
      document.getElementById('lb-book-label').textContent = card.dataset.bookLabel;
      lb.classList.remove('hidden');
      lb.classList.add('flex');
      document.body.style.overflow = 'hidden';
      document.getElementById('gallery-lightbox-close').focus();
    };
    var closeBox = function () {
      lb.classList.add('hidden');
      lb.classList.remove('flex');
      document.body.style.overflow = '';
    };
    document.addEventListener('click', function (e) {
      var card = e.target.closest('[data-gallery-card]');
      if (card) { openBox(card); return; }
      if (e.target.closest('#gallery-lightbox-close')) { closeBox(); return; }
      if (e.target === lb) closeBox();
    });
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' && !lb.classList.contains('hidden')) closeBox();
    });
  }

  /* 4. FAQ-акордеон */
  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-faq-toggle]');
    if (!btn) return;
    var panel = document.getElementById(btn.dataset.faqTarget);
    if (!panel) return;
    var expanded = btn.getAttribute('aria-expanded') === 'true';
    btn.setAttribute('aria-expanded', String(!expanded));
    panel.classList.toggle('hidden');
    var chevron = btn.querySelector('svg');
    if (chevron) chevron.classList.toggle('rotate-180');
  });

  /* 5. Плавний скрол до якоря при завантаженні (1:1 HomePage.tsx) */
  if (window.location.hash) {
    var el = document.getElementById(window.location.hash.slice(1));
    if (el) setTimeout(function () { el.scrollIntoView({ behavior: 'smooth' }); }, 150);
  }
})();
