"""Email-ваучери бронювань.

def _week_bounds(offset=0):
    today = timezone.localdate()
    monday = today - timedelta(days=today.weekday()) + timedelta(weeks=offset)
    return monday, monday + timedelta(days=6)


def kpi_context():
    today = timezone.localdate()
    active = Booking.objects.exclude(status="cancelled")
    today_qs = active.filter(date=today).order_by("time")
    pending = Booking.objects.filter(status="pending_payment")
    monday, sunday = _week_bounds(0)
    week = active.filter(date__gte=monday, date__lte=sunday)
    revenue = (
        week.exclude(status="pending_payment").aggregate(s=Sum("total_price"))["s"] or 0
    )
    booked = week.aggregate(s=Sum("boards_total"))["s"] or 0
    capacity = len(ALL_SLOTS) * 7 * MAX_BOARDS_PER_SLOT
    return {
        "today": today,
        "today_count": today_qs.count(),
        "today_boards": today_qs.aggregate(s=Sum("boards_total"))["s"] or 0,
        "upcoming": today_qs[:3],
        "pending_count": pending.count(),
        "pending_sum": pending.aggregate(s=Sum("total_price"))["s"] or 0,
        "week_revenue": revenue,
        "occupancy": round(booked / capacity * 100) if capacity else 0,
    }


def occupancy_context(offset=0):
    monday, sunday = _week_bounds(offset)
    days = [monday + timedelta(days=i) for i in range(7)]
    data = (
        Booking.objects.exclude(status="cancelled")
        .filter(date__gte=monday, date__lte=sunday)
        .values("date", "time")
        .annotate(total=Sum("boards_total"))
    )
    lookup = {(d["date"], d["time"].strftime("%H:%M")): d["total"] for d in data}
    rows = [
        {"time": s, "cells": [lookup.get((day, s), 0) for day in days]}
        for s in ALL_SLOTS
    ]
    return {
        "days": days,
        "rows": rows,
        "day_totals": [sum(r["cells"][i] for r in rows) for i in range(7)],
        "week_offset": offset,
        "week_label": f"{monday.strftime('%d.%m')} — {sunday.strftime('%d.%m')}",
        "today": timezone.localdate(),
    }


KANBAN_META = {
    "pending_payment": ("Pending Bizum", "text-amber-400", "bg-amber-400"),
    "confirmed": ("Confirmed", "text-sky-400", "bg-sky-400"),

def kanban_columns():

                "text_class": text_c,

def _pipeline_response(request):

    ctx.update(kpi_context())
    return render(request, "dashboard/partials/_pipeline_response.html", ctx)

class BookingStatusUpdateView(StaffRequiredMixin, View):

"""Telegram-бот власника: /today, /confirm TY-XXXX, /delivered, /complete, /stats.

Отримує оновлення через webhook (POST /telegram/webhook/).
Безпека: відповідає лише в Telegram-чат із TELEGRAM_CHAT_ID (.env).
"""
import json
import os
import urllib.request

from django.http import JsonResponse
from django.utils import timezone
from django.utils.decorators import method_decorator
from django.views import View
from django.views.decorators.csrf import csrf_exempt

from bookings.models import Booking

HELP_TEXT = (
    "🏄 Команди TUYYO Bot:\n"
    "/today — бронювання на сьогодні\n"
    "/confirm TY-XXXX — підтвердити оплату\n"
    "/delivered TY-XXXX — видано на пляжі\n"
    "/complete TY-XXXX — завершити (франшиза повернута)\n"
    "/stats — коротка статистика"
)

_SET_STATUS = {
    "/confirm": "confirmed",
    "/delivered": "delivered",
    "/complete": "completed",
}


def _fmt_booking(b):
    return (
        f"#{b.reference} · {b.date} {b.time:%H:%M}\n"
        f"  👤 {b.full_name} ({b.phone})\n"
        f"  📍 {b.location.name} · {b.boards_total} дощ. · €{b.total_price} · {b.payment_method}\n"
        f"  📌 {b.status}"
    )


def handle_command(text: str) -> str:
    """Чиста логіка команд — тестується без мережі."""
    parts = text.strip().split()
    if not parts:
        return HELP_TEXT
    cmd = parts[0].lower().split("@")[0]  # прибираємо @botname

    if cmd in ("/start", "/help"):
        return HELP_TEXT

    if cmd == "/today":
        qs = (
            Booking.objects.filter(date=timezone.localdate())
            .exclude(status="cancelled")
            .order_by("time")
            .select_related("location")
        )
        if not qs:
            return "🏖 Сьогодні бронювань немає."
        return "📅 Бронювання на сьогодні:\n\n" + "\n\n".join(_fmt_booking(b) for b in qs)

    if cmd in _SET_STATUS:
        if len(parts) < 2:
            return f"Вкажіть референс: {cmd} TY-1234"
        ref = parts[1].upper()
        booking = Booking.objects.filter(reference=ref).first()
        if not booking:
            return f"❌ {ref} не знайдено"
        booking.status = _SET_STATUS[cmd]
        booking.save(update_fields=["status"])
        return f"✅ {ref} → {booking.status}"

    if cmd == "/stats":
        today = timezone.localdate()
        active = Booking.objects.exclude(status="cancelled")
        pending = Booking.objects.filter(status="pending_payment")
        return (
            "📊 Статистика TUYYO:\n"
            f"• сьогодні: {active.filter(date=today).count()}\n"
            f"• очікують оплати: {pending.count()} (€{pending.count() and sum(b.total_price for b in pending) or 0})\n"
            f"• всього активних: {active.count()}"
        )

    return "Невідома команда. /help"


def _send_telegram(token: str, chat_id, text: str):
    payload = json.dumps({"chat_id": chat_id, "text": text}).encode("utf-8")
    req = urllib.request.Request(
        f"https://api.telegram.org/bot{token}/sendMessage",
        data=payload,
        headers={"Content-Type": "application/json"},
    )
    urllib.request.urlopen(req, timeout=5)


@method_decorator(csrf_exempt, name="dispatch")
class TelegramWebhookView(View):
    """Приймає оновлення від Telegram (fail-safe: нічого не ламає сайт)."""

    def get(self, request):
        return JsonResponse({"ok": True, "service": "tuyyo-telegram-webhook"})

    def post(self, request):
        try:
            data = json.loads(request.body or b"{}")
        except json.JSONDecodeError:
            return JsonResponse({"ok": False, "error": "bad json"}, status=400)
        message = data.get("message") or {}
        text = message.get("text") or ""
        chat_id = (message.get("chat") or {}).get("id")
        token = os.environ.get("TELEGRAM_BOT_TOKEN", "")
        allowed_chat = os.environ.get("TELEGRAM_CHAT_ID", "")
        # Відповідаємо лише власнику; без токену — тихо ігноруємо
        if not (token and allowed_chat and str(chat_id) == allowed_chat):
            return JsonResponse({"ok": True})
        try:
            _send_telegram(token, chat_id, handle_command(text))
        except Exception:
            pass
        return JsonResponse({"ok": True})
EOF

"""Реєстрація Telegram-webhook: SITE_URL + /telegram/webhook/."""
import json
import os
import urllib.request

from django.conf import settings
from django.core.management.base import BaseCommand


class Command(BaseCommand):
    help = "Встановлює Telegram-webhook для бота власника"

    def handle(self, *args, **options):
        token = os.environ.get("TELEGRAM_BOT_TOKEN", "")
        if not token:
            self.stderr.write(self.style.ERROR("TELEGRAM_BOT_TOKEN не задано у .env"))
            return
        url = f"{settings.SITE_URL.rstrip('/')}/telegram/webhook/"
        payload = json.dumps({"url": url}).encode("utf-8")
        req = urllib.request.Request(
            f"https://api.telegram.org/bot{token}/setWebhook",
            data=payload,
            headers={"Content-Type": "application/json"},
        )
        with urllib.request.urlopen(req, timeout=10) as resp:
            result = json.loads(resp.read().decode("utf-8"))
        if result.get("ok"):
            self.stdout.write(self.style.SUCCESS(f"✅ Webhook встановлено: {url}"))
        else:
            self.stderr.write(self.style.ERROR(f"❌ Помилка Telegram: {result}"))
EOF

        full_name="Test Client",
        phone="+34 600 000 000",
        id_passport="X0000000",
        payment_method="bizum",
        boards_total=2,
        unit_price=40,
        subtotal=80,
        delivery_fee=0,
        total_price=80,
        deposit_amount=150,
        status="pending_payment",
    )
    defaults.update(overrides)
    return Booking.objects.create(**defaults)

class KanbanTests(TestCase):
    @classmethod
    def setUpTestData(cls):
        seed()
        cls.staff = User.objects.create_user("owner", "o@t.c", "pass12345", is_staff=True)

    def setUp(self):
        self.client.force_login(self.staff)

    def test_columns_render_bookings(self):
        b = make_booking(full_name="Kanban Visible")
        html = self.client.get("/dashboard/").content.decode()
        self.assertIn(b.reference, html)
        self.assertIn("Kanban Visible", html)

    def test_status_update_valid_transition(self):
        b = make_booking(status="pending_payment")
        r = self.client.post(f"/dashboard/bookings/{b.pk}/status/", {"status": "confirmed"},
                             HTTP_HX_REQUEST="true")
        self.assertEqual(r.status_code, 200)
        b.refresh_from_db()
        self.assertEqual(b.status, "confirmed")

    def test_status_update_invalid_transition_ignored(self):
        b = make_booking(status="pending_payment")
        self.client.post(f"/dashboard/bookings/{b.pk}/status/", {"status": "completed"},
                         HTTP_HX_REQUEST="true")
        b.refresh_from_db()
        self.assertEqual(b.status, "pending_payment")  # перехід заборонено

    def test_resend_voucher_sends_email_and_marks(self):
        b = make_booking(email="client@example.com")
        self.client.post(f"/dashboard/bookings/{b.pk}/resend-voucher/", {},
